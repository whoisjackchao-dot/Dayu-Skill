# dayu-viz M0–M1 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立一个纯解析层（Markdown → 标准化 `DayuProject` 模型）与最小可用的 V1 驾驶舱视图，指向任意 dayu 项目的 `docs/` 即可端到端看到该项目的阶段、当前节点、阻塞、待批准与交接状态，且文档一变页面自动跟随。

**Architecture:** 三层严格分离——`src/core/**` 为纯函数解析层（零 `node:*` IO 依赖、无文件系统访问、无时钟），`src/server/**` 是唯一 IO 层（`node:http` + `fs.watch` + SSE），`web/**` 只消费 `/api/model` 返回的 JSON，不含任何解析逻辑。后端零 npm 依赖，前端零构建步骤（浏览器原生 ES Module）。

**Tech Stack:** Node ≥ 20（开发机实测 v24.16.0）、`node:http`、`node:fs`、`node:test` + `node:assert`、原生 ES Module、无打包器、无运行时 npm 依赖。

**Spec:** `docs/superpowers/specs/2026-10-11-dayu-viz-dashboard-design.md`

## Global Constraints

- **项目根**：`D:\DSH\Dayu-Skill\dayu-viz\`（git 仓库内，唯一权威副本）。所有路径在下文均相对该根。
- **零运行时依赖**：`package.json` 的 `dependencies` 必须为空。不得引入 express / vite / react / echarts 之外任何东西；M1 **完全不引入**图表库。
- **分层铁律**：`src/core/**` 中不得出现 `import ... from 'node:fs'`、`node:http`、`node:child_process`、`Date.now()`、`new Date()`。解析结果必须完全由输入文本决定。
- **绝不编造**：任何读不到的字段取值 `'unknown'`，不得用 `0`、`''`、`null` 或默认值掩盖。UI 显示为"未提供"。
- **状态枚举固定**（不得自创）：用例 `NOT_RUN` / `BLOCKED` / `FAIL` / `PASS` / `NOT_APPLICABLE_APPROVED` / `unknown`；决策 `approved` / `pending` / `undecided` / `unknown`。
- **配色语义**：`BLOCKED` 与 `NOT_RUN` 在任何情况下**不得**渲染为绿色或任何表示成功的样式。
- **文件写入限制（实测环境约束）**：`D:\DSH` 位于 **exFAT** 卷，不支持硬链接。全量新建文件的原子写工具会失败（`EISDIR ... link`）。**新建文件**需先写到 `C:\Users\home\_dsh_stage\` 再用 `Copy-Item` 拷入项目；**修改已有文件**用原地编辑即可。
- **提交纪律**：每个 Task 结束提交一次。**只 commit，不 push**——推送是 `Q01`，未获用户授权不得执行。
- **node 命令**：本机 `node` 在 `C:\Program Files\nodejs\node.exe`（v24.16.0）。若 `node` 不在 PATH，用绝对路径。

```js
// 纯解析：零 node:* 依赖、无 IO、无时钟。所有输出仅由输入文本决定。

export function toRe(pattern) {
  if (pattern instanceof RegExp) return pattern;
  return new RegExp(String(pattern).replace(/[.*+?^${}()|[\]\\]/g, '\\$&'));
}

export function splitRow(line) {
  const t = String(line ?? '').trim();
  if (!t.startsWith('|')) return null;
  const body = t.replace(/^\|/, '').replace(/\|$/, '');
  return body.split('|').map((c) => c.trim());
}

export function isSeparatorRow(cells) {
  if (!Array.isArray(cells) || cells.length === 0) return false;
  return cells.every((c) => /^:?-{2,}:?$/.test(String(c).replace(/\s/g, '')));
}

export function parseDocument(text) {
  const lines = String(text ?? '').split(/\r?\n/);
  const headings = [];
  const tables = [];
  const stack = [];

  for (let i = 0; i < lines.length; i++) {
    const h = /^(#{1,6})\s+(.*)$/.exec(lines[i]);
    if (h) {
      const level = h[1].length;
      const htext = h[2].trim();
      while (stack.length && stack[stack.length - 1].level >= level) stack.pop();
      stack.push({ level, text: htext, line: i + 1 });
      headings.push({ level, text: htext, line: i + 1 });
      continue;
    }

    const cells = splitRow(lines[i]);
    if (!cells) continue;
    const next = splitRow(lines[i + 1] ?? '');
    if (!next || !isSeparatorRow(next)) continue;

    const rows = [];
    let j = i + 2;
    while (j < lines.length) {
      const r = splitRow(lines[j]);
      if (!r) break;
      rows.push(r);
      j++;
    }
    tables.push({
      header: cells,
      rows,
      line: i + 1,
      headings: stack.map((s) => s.text),
    });
    i = j - 1;
  }

  return { lines, headings, tables };
}

export function findTables(doc, pattern) {
  const re = toRe(pattern);
  return doc.tables.filter((t) => t.headings.some((h) => re.test(h)));
}

export function colIndex(header, ...names) {
  for (const name of names) {
    const re = toRe(name);
    const i = header.findIndex((h) => re.test(h));
    if (i >= 0) return i;
  }
  return -1;
}

export function cellAt(row, idx) {
  if (!Array.isArray(row) || idx < 0 || idx >= row.length) return null;
  const v = row[idx];
  return v === undefined ? null : v;
}

export function hasPlaceholder(s) {
  return /<[^<>\n]{1,60}>/.test(String(s ?? ''));
}

export function isBlank(s) {
  return s === undefined || s === null || String(s).trim() === '';
}

export function checkboxItems(text) {
  const out = [];
  for (const line of String(text ?? '').split(/\r?\n/)) {
    const m = /^\s*-\s*\[( |x|X)\]\s*(.*)$/.exec(line);
    if (m) out.push({ checked: m[1].toLowerCase() === 'x', text: m[2].trim() });
  }
  return out;
}
```

```js
export const UNKNOWN = 'unknown';

const CASE_STATES = [
  'NOT_APPLICABLE_APPROVED', 'NOT_RUN', 'BLOCKED', 'FAIL', 'PASS',
];

export function emptyProject(root, docsDir, parsedAt) {
  return {
    meta: { name: UNKNOWN, root, docsDir, parsedAt, inventory: [], warnings: [] },
    status: {
      phase: UNKNOWN,
      currentNode: UNKNOWN,
      sessionGoal: UNKNOWN,
      docVersion: UNKNOWN,
      handover: {
        updatedAt: UNKNOWN, completed: UNKNOWN, notCompleted: UNKNOWN,
        next: UNKNOWN, awaitingUser: UNKNOWN, dontReask: UNKNOWN, untrusted: UNKNOWN,
      },
    },
    requirements: [],
    gates: [],
    decisions: [],
    contracts: [],
    milestones: [],
    cases: [],
    trace: [],
    evidence: [],
    gaps: [],
    experiments: [],
    acceptance: [],
    redflags: [],
    agents: { sessionId: UNKNOWN, members: [], tasks: [], messages: [], subagents: [], stats: {} },
  };
}

export function addWarning(project, code, message, ref) {
  project.meta.warnings.push(ref ? { code, message, ref } : { code, message });
}

export function stripEmphasis(s) {
  return String(s ?? '').replace(/\*\*/g, '').replace(/`/g, '').trim();
}

export function normalizeDecisionState(text) {
  const t = stripEmphasis(text);
  if (!t) return UNKNOWN;
  if (t.includes('已批准')) return 'approved';
  if (t.includes('待批准')) return 'pending';
  if (t.includes('未决')) return 'undecided';
  return UNKNOWN;
}

export function normalizeCaseState(text) {
  const t = stripEmphasis(text).toUpperCase();
  if (!t) return UNKNOWN;
  // NOT_APPLICABLE_APPROVED 必须先匹配，否则会被 NOT_RUN 的包含关系误判
  for (const s of CASE_STATES) {
    if (t.includes(s)) return s;
  }
  return UNKNOWN;
}

export function extractIds(text, prefix) {
  const re = new RegExp(`\\b${prefix}(\\d{1,3})\\b`, 'g');
  const seen = new Set();
  const out = [];
  for (const m of String(text ?? '').matchAll(re)) {
    const id = `${prefix}${m[1]}`;
    if (!seen.has(id)) { seen.add(id); out.push(id); }
  }
  return out;
}

export function extractDate(text) {
  const m = /(\d{4}-\d{2}-\d{2})/.exec(String(text ?? ''));
  return m ? m[1] : null;
}
```

> 注意 `normalizeCaseState` 的匹配顺序：`NOT_APPLICABLE_APPROVED` 含 `APPROVED`，`NOT_RUN` 与 `BLOCKED` 无包含关系，但把最长的放前面是必要防御。

```js
import { parseDocument, hasPlaceholder } from './markdown.mjs';

/** 启发式阈值：占位符数量达到此值即认为该文档仍是未填模板。 */
export const TEMPLATE_PLACEHOLDER_THRESHOLD = 5;

export function buildInventory(entries) {
  return (entries ?? []).map((entry) => {
    const text = entry.text ?? '';
    if (!entry.exists) {
      return {
        file: entry.file, exists: false, isTemplate: false,
        placeholders: 0, dataRows: 0, bytes: 0, mtime: null,
      };
    }
    const doc = parseDocument(text);
    let placeholders = 0;
    for (const line of doc.lines) if (hasPlaceholder(line)) placeholders++;
    const dataRows = doc.tables.reduce((n, t) => n + t.rows.length, 0);
    return {
      file: entry.file,
      exists: true,
      isTemplate: placeholders >= TEMPLATE_PLACEHOLDER_THRESHOLD,
      placeholders,
      dataRows,
      bytes: entry.bytes ?? Buffer.byteLength(text, 'utf8'),
      mtime: entry.mtime ?? null,
    };
  });
}
```

> `Buffer` 是 Node 全局，但本模块只用它做字节计数（纯计算），不构成 IO。若在纯浏览器环境中复用此模块，需替换为 `new TextEncoder().encode(text).length`。

```js
import { parseDocument, findTables, colIndex, cellAt } from './markdown.mjs';
import { UNKNOWN, addWarning, extractIds, extractDate, stripEmphasis } from './model.mjs';

const HANDOVER_KEYS = [
  ['updatedAt', /最后更新/],
  ['completed', /本次完成/],
  ['notCompleted', /本次未完成/],
  ['next', /下一步/],
  ['awaitingUser', /待用户决定/],
  ['dontReask', /不要重复询问/],
  ['untrusted', /已知不可信/],
];

/** 就地写入 project.status。 */
export function parseIndex(text, project, file = '00-index.md') {
  const doc = parseDocument(text);
  const status = project.status;

  // 1. 当前状态表：按标题路径定位，不按位置
  // 叠加表头判据：findTables 的标题路径包含文档标题本身，只按标题匹配可能命中文档里的第一张表。
  const t = findTables(doc, /当前状态/)
    .find((x) => colIndex(x.header, '字段') >= 0 && colIndex(x.header, '值') >= 0);
  if (!t) {
    addWarning(project, 'NO_STATUS_TABLE', '未找到含 字段/值 列的「当前状态」表', { file });
    return;
  }
  const kIdx = colIndex(t.header, '字段');
  const vIdx = colIndex(t.header, '值');

  for (let i = 0; i < t.rows.length; i++) {
    const key = stripEmphasis(cellAt(t.rows[i], kIdx) ?? '');
    const val = (cellAt(t.rows[i], vIdx) ?? '').trim();
    const line = t.line + 2 + i;

    if (/当前阶段/.test(key)) {
      const m = /阶段\s*([0-6])/.exec(val);
      // 带占位符的「阶段 <0–6>」不构成有效阶段
      status.phase = m && !val.includes('<') ? Number(m[1]) : UNKNOWN;
    } else if (/当前节点/.test(key)) {
      const ids = extractIds(val, 'M');
      if (ids.length === 0) {
        status.currentNode = UNKNOWN;
      } else {
        status.currentNode = ids[0];
        if (ids.length > 1) {
          addWarning(
            project,
            'MULTI_NODE',
            `当前节点表述含多个里程碑（${ids.join('、')}），取首个 ${ids[0]}`,
            { file, line },
          );
        }
      }
    } else if (/本会话目标/.test(key)) {
      status.sessionGoal = val || UNKNOWN;
    } else if (/文档版本/.test(key)) {
      status.docVersion = val || UNKNOWN;
    }
  }

  // 2. 交接块：按「键：值」行解析，不依赖固定顺序
  const handoverHeading = doc.headings.find((h) => /交接/.test(h.text));
  if (!handoverHeading) {
    addWarning(project, 'NO_HANDOVER', '未找到交接块', { file });
    return;
  }
  status.handover.updatedAt = extractDate(handoverHeading.text) ?? UNKNOWN;

  const start = handoverHeading.line; // 1-based，下一行开始
  for (let i = start; i < doc.lines.length; i++) {
    const line = doc.lines[i];
    if (/^#{1,6}\s/.test(line)) break;
    const m = /^\s*-\s*\*\*(.+?)\*\*[：:]\s*(.*)$/.exec(line);
    if (!m) continue;
    const label = m[1].trim();
    const value = m[2].trim();
    const hit = HANDOVER_KEYS.find(([, re]) => re.test(label));
    if (hit && hit[0] !== 'updatedAt' && value) {
      status.handover[hit[0]] = value;
    }
  }
}
```

```js
import { parseDocument, findTables, colIndex, cellAt } from './markdown.mjs';
import {
  UNKNOWN, addWarning, normalizeDecisionState, extractIds, extractDate, stripEmphasis,
} from './model.mjs';

/** 就地写入 project.decisions。 */
export function parseDecisions(text, project, file = '01-decisions.md') {
  const doc = parseDocument(text);

  // 注意：findTables 把文档标题也算进标题路径，而本文件标题就是「决策登记」，
  // 只按标题匹配会命中所有表。因此必须叠加表头判据。
  const table = doc.tables.find(
    (t) => colIndex(t.header, 'ID') >= 0 && colIndex(t.header, '决策内容') >= 0,
  );
  if (!table) {
    addWarning(project, 'NO_DECISION_TABLE', '未找到决策表', { file });
    return;
  }

  const idIdx = colIndex(table.header, 'ID');
  const qIdx = colIndex(table.header, '决策内容');
  const sIdx = colIndex(table.header, '当前依据/状态', '状态', '依据');
  const nIdx = colIndex(table.header, '影响节点');

  if (idIdx < 0 || qIdx < 0) {
    addWarning(project, 'BAD_DECISION_TABLE', '决策表缺少 ID 或 决策内容 列', { file, line: table.line });
    return;
  }

  // 先扫批准记录，建立 id -> {notCovering, note}
  const approvals = collectApprovals(doc);

  table.rows.forEach((row, i) => {
    const id = stripEmphasis(cellAt(row, idIdx) ?? '');
    if (!/^Q\d+$/.test(id)) return;
    const stateText = cellAt(row, sIdx) ?? '';
    const approval = approvals.get(id) ?? { notCovering: [], note: null };
    project.decisions.push({
      id,
      question: (cellAt(row, qIdx) ?? '').trim(),
      state: normalizeDecisionState(stateText),
      date: extractDate(stateText),
      basis: stripEmphasis(stateText),
      affectedNodes: extractIds(cellAt(row, nIdx) ?? '', 'M'),
      approvalNote: approval.note,
      notCovering: approval.notCovering,
      source: { file, line: table.line + 2 + i },
    });
  });
}

function collectApprovals(doc) {
  const map = new Map();
  let current = null;
  for (const line of doc.lines) {
    const h = /^###\s+(Q\d+)\s*批准依据/.exec(line);
    if (h) {
      current = { id: h[1], notCovering: [], note: null, body: [] };
      map.set(current.id, current);
      continue;
    }
    if (/^#{1,6}\s/.test(line) && !h) { current = null; continue; }
    if (!current) continue;
    current.body.push(line);
    const m = /^\*\*本批准不覆盖\*\*[：:]\s*(.+)$/.exec(line.trim());
    if (m) current.note = m[1].trim();
  }
  // 把 note 按「；」拆成条目，保留原文措辞
  for (const a of map.values()) {
    a.notCovering = a.note
      ? a.note.split('；').map((s) => s.trim()).filter(Boolean)
      : [];
  }
  return map;
}
```

```js
import { parseDocument, colIndex, cellAt } from './markdown.mjs';
import { UNKNOWN, addWarning, stripEmphasis } from './model.mjs';

/** 就地写入 project.requirements 与 project.gates。 */
export function parseRequirements(text, project, file = '02-requirements.md') {
  const doc = parseDocument(text);

  // 注意：findTables 把文档标题也算进标题路径，本文件标题含「验收门槛」，
  // 因此门槛表的定位必须叠加表头判据，否则会命中文档里的第一张表。
  const byHeader = (...names) => doc.tables.find((t) => names.every((n) => colIndex(t.header, n) >= 0));

  // ---- 需求表 ----
  const reqTable = byHeader('编号', '已确认要求') ?? byHeader('ID', '已确认要求');
  if (reqTable) {
    const idIdx = colIndex(reqTable.header, '编号', 'ID');
    const txtIdx = colIndex(reqTable.header, '已确认要求', '要求', '内容');
    if (idIdx >= 0 && txtIdx >= 0) {
      reqTable.rows.forEach((row, i) => {
        const id = stripEmphasis(cellAt(row, idIdx) ?? '');
        if (!/^R\d+$/.test(id)) return;
        project.requirements.push({
          id,
          text: (cellAt(row, txtIdx) ?? '').trim(),
          gates: [],
          source: { file, line: reqTable.line + 2 + i },
        });
      });
    } else {
      addWarning(project, 'BAD_REQUIREMENT_TABLE', '需求表缺少 编号 或 已确认要求 列', { file, line: reqTable.line });
    }
  } else {
    addWarning(project, 'NO_REQUIREMENT_TABLE', '未找到需求表', { file });
  }

  // ---- 验收门槛表：两种真实形态 ----
  const gateTable = byHeader('门槛', '判定标准') ?? byHeader('门槛', '数值');
  if (gateTable) {
    const idIdx = colIndex(gateTable.header, '门槛', 'ID');
    const critIdx = colIndex(gateTable.header, '判定标准');
    const topicIdx = colIndex(gateTable.header, '主题');
    const valIdx = colIndex(gateTable.header, '数值', '值');
    gateTable.rows.forEach((row, i) => {
      const id = stripEmphasis(cellAt(row, idIdx) ?? '');
      if (!/^G\d+/.test(id)) return;
      const rawCrit = critIdx >= 0 ? (cellAt(row, critIdx) ?? '').trim() : '';
      const rawTopic = topicIdx >= 0 ? (cellAt(row, topicIdx) ?? '').trim() : '';
      const rawVal = valIdx >= 0 ? stripEmphasis(cellAt(row, valIdx) ?? '').trim() : '';
      let state = 'defined';
      if (critIdx >= 0 && rawCrit) state = 'defined';
      else if (/^TBD$/i.test(rawVal) || rawVal === '') state = 'tbd';
      else if (rawVal.includes('待') || rawVal.includes('未定')) state = 'pending';
      else state = 'defined';
      project.gates.push({
        id,
        topic: rawTopic || UNKNOWN,
        value: critIdx >= 0 && rawCrit ? rawCrit : (rawVal || UNKNOWN),
        state,
        source: { file, line: gateTable.line + 2 + i },
      });
    });
  } else {
    addWarning(project, 'NO_GATE_TABLE', '未找到验收门槛表', { file });
  }
}
```

```js
      if (!milestone) {
        // 卡片存在但节点序列里没有 —— 登记偏差而不是静默忽略
        addWarning(project, 'CARD_WITHOUT_NODE', `里程碑卡片 ${id} 未出现在节点序列表中`, { file, line: i + 1 });
        current = null;
        continue;
      }
      milestone.name = h[2].trim() || milestone.name;
      milestone.fields = emptyFields();
      current = milestone;
      continue;
    }
    if (/^#{1,6}\s/.test(line) && !h) {
      if (/里程碑卡片/.test(line) === false && current) {
        // 卡片区段结束（遇到同级或更高级标题）
        if (/^#{1,3}\s/.test(line) && !/^####/.test(line)) current = null;
      }
      continue;
    }
    if (!current) continue;
    const m = /^\s*-\s*\*\*(.+?)\*\*[：:]\s*(.*)$/.exec(line);
    if (!m) continue;
    const label = m[1].trim();
    const value = m[2].trim();
    const hit = FIELD_KEYS.find(([, re]) => re.test(label));
    if (!hit) continue;
    const [key] = hit;
    if (LIST_FIELDS.has(key)) {
      current.fields[key] = splitItems(value);
    } else if (key === 'cases') {
      current.fields.cases = extractCaseIds(value);
    } else {
      current.fields[key] = value || UNKNOWN;
    }
  }

  for (const m of project.milestones) {
    if (m.fields.notDone.length === 0) {
      addWarning(project, 'MISSING_NOT_DONE', `里程碑 ${m.id} 缺少「不算完成」字段`, { file, line: m.source.line });
    }
  }

  // ---- 3. DoR ----
  // 文档标题是「里程碑与 Definition of Ready」，只按标题匹配会命中所有表；按表头定位。
  const dorTable = doc.tables.find(
    (t) => colIndex(t.header, '条件') >= 0 && colIndex(t.header, '状态') >= 0,
  );
  if (dorTable) {
    const nIdx = colIndex(dorTable.header, '#', '序号');
    const cIdx = colIndex(dorTable.header, '条件');
    const sIdx = colIndex(dorTable.header, '状态');
    dorTable.rows.forEach((row) => {
      project.planExtras.dor.push({
        n: (cellAt(row, nIdx) ?? '').trim(),
        condition: (cellAt(row, cIdx) ?? '').trim(),
        state: (cellAt(row, sIdx) ?? '').trim() || UNKNOWN,
      });
    });
  }

  // ---- 4. 常量批准前置 ----
  const constTable = doc.tables.find(
    (t) => colIndex(t.header, '符号') >= 0 && colIndex(t.header, '批准值') >= 0,
  );
  if (constTable) {
    const symIdx = colIndex(constTable.header, '符号');
    const valIdx = colIndex(constTable.header, '批准值');
    const unitIdx = colIndex(constTable.header, '单位');
    const basisIdx = colIndex(constTable.header, '批准依据');
    constTable.rows.forEach((row) => {
      const symbol = stripEmphasis(cellAt(row, symIdx) ?? '');
      if (!symbol) return;
      project.planExtras.constants.push({
        symbol,
        value: stripEmphasis(cellAt(row, valIdx) ?? '') || UNKNOWN,
        unit: (cellAt(row, unitIdx) ?? '').trim() || UNKNOWN,
        basis: stripEmphasis(cellAt(row, basisIdx) ?? '') || UNKNOWN,
      });
    });
  }
}

/** 把「① a；② b」或「a、b」拆成条目。 */
function splitItems(value) {
  return String(value ?? '')
    .split(/[；;]/)
    .flatMap((s) => s.split(/(?=[①②③④⑤⑥⑦⑧⑨⑩])/))
    .map((s) => s.replace(/^[①②③④⑤⑥⑦⑧⑨⑩\s]+/, '').trim())
    .filter(Boolean);
}

const CASE_ID_RE = /\b([A-Z]{2,6})-(\d{1,3})\b/g;

/** 抽取用例 ID（域前缀-两位序号），并展开 `IMG-01…04` 这类闭区间写法。 */
function extractCaseIds(value) {
  const out = [];
  const seen = new Set();
  const push = (id) => { if (!seen.has(id)) { seen.add(id); out.push(id); } };
  for (const m of String(value ?? '').matchAll(CASE_ID_RE)) {
    push(`${m[1]}-${m[2]}`);
  }
  // 处理 `IMG-01…04` / `TUI-01`—`TUI-18` 的区间写法
  const rangeRe = /\b([A-Z]{2,6})-(\d{1,3})\s*[…—–\-~至到]+\s*(\d{1,3})\b/g;
  for (const m of String(value ?? '').matchAll(rangeRe)) {
    const from = Number(m[2]);
    const to = Number(m[3]);
    if (to >= from && to - from <= 30) {
      for (let n = from; n <= to; n++) push(`${m[1]}-${String(n).padStart(2, '0')}`);
    }
  }
  return out;
}
```

> `extractCaseIds` 的区间展开上限为 30 条，避免 `M1`—`M6` 之类误匹配产生爆炸。`extractIds` 在 `model.mjs` 里，用于抽 `M` 编号。

```js
// 唯一读盘处。core 层不得读文件，所有 IO 集中在此。
import fs from 'node:fs/promises';
import path from 'node:path';
import { emptyProject, UNKNOWN, addWarning } from '../core/model.mjs';
import { buildInventory } from '../core/parse-inventory.mjs';
import { parseIndex } from '../core/parse-index.mjs';
import { parseDecisions } from '../core/parse-decisions.mjs';
import { parseRequirements } from '../core/parse-requirements.mjs';
import { parsePlan } from '../core/parse-plan.mjs';

export const DOC_FILES = [
  '00-index.md',
  '01-decisions.md',
  '02-requirements.md',
  '03-feasibility.md',
  '04-adr.md',
  '05-plan.md',
  '06-test-matrix.md',
  '07-evidence-log.md',
  '08-readiness.md',
  '09-acceptance.md',
  '10-delivery.md',
  'contracts/README.md',
];

export async function readDocs(docsDir) {
  const entries = [];
  for (const file of DOC_FILES) {
    const abs = path.join(docsDir, file);
    try {
      const stat = await fs.stat(abs);
      const text = await fs.readFile(abs, 'utf8');
      entries.push({ file, exists: true, text, bytes: stat.size, mtime: stat.mtime.toISOString() });
    } catch (err) {
      if (err.code === 'ENOENT') {
        entries.push({ file, exists: false });
      } else {
        throw err;
      }
    }
  }
  return entries;
}

export function buildProject(docsDir, entries, { parsedAt, projectRoot }) {
  const project = emptyProject(projectRoot ?? path.dirname(docsDir), docsDir, parsedAt);
  project.meta.inventory = buildInventory(entries);

  const byFile = new Map(entries.map((e) => [e.file, e]));
  const get = (f) => byFile.get(f)?.text;

  const idx = get('00-index.md');
  if (idx) {
    parseIndex(idx, project, '00-index.md');
    const title = /^#\s+(.+)$/m.exec(idx);
    if (title) project.meta.name = title[1].trim();
  } else {
    addWarning(project, 'NO_INDEX', '缺少 00-index.md，无法确定项目阶段');
  }

  const dec = get('01-decisions.md');
  if (dec) parseDecisions(dec, project, '01-decisions.md');

  const req = get('02-requirements.md');
  if (req) parseRequirements(req, project, '02-requirements.md');

  const plan = get('05-plan.md');
  if (plan) parsePlan(plan, project, '05-plan.md');

  return project;
}

export async function loadProject(projectRoot) {
  const docsDir = path.join(projectRoot, 'docs');
  let stat;
  try {
    stat = await fs.stat(docsDir);
  } catch {
    throw new Error(`docs 目录不存在：${docsDir}`);
  }
  if (!stat.isDirectory()) throw new Error(`不是目录：${docsDir}`);

  const entries = await readDocs(docsDir);
  const parsedAt = new Date().toISOString();
  return buildProject(docsDir, entries, { parsedAt, projectRoot });
}
```

```js
import fs from 'node:fs';

/** 递归监听多个目录，去抖后回调一次。返回 { close() }。 */
export function createWatcher(paths, onChange, { debounceMs = 300 } = {}) {
  const watchers = [];
  let timer = null;

  const fire = () => {
    if (timer) clearTimeout(timer);
    timer = setTimeout(() => {
      timer = null;
      try { onChange(); } catch { /* 回调异常不得杀死监听 */ }
    }, debounceMs);
  };

  for (const p of paths ?? []) {
    try {
      const w = fs.watch(p, { recursive: true }, fire);
      w.on('error', () => { /* 监听失败降级为手动刷新，不抛错 */ });
      watchers.push(w);
    } catch {
      // 目录不存在或平台不支持递归监听：降级，不抛错
    }
  }

  return {
    close() {
      if (timer) clearTimeout(timer);
      timer = null;
      for (const w of watchers) { try { w.close(); } catch { /* ignore */ } }
      watchers.length = 0;
    },
  };
}
```

- [ ] **Step 4: 实现 `src/server/server.mjs`**

```js
import http from 'node:http';
import fs from 'node:fs/promises';
import path from 'node:path';
import { loadProject } from './project.mjs';

const MIME = {
  '.html': 'text/html; charset=utf-8',
  '.mjs': 'text/javascript; charset=utf-8',
  '.js': 'text/javascript; charset=utf-8',
  '.css': 'text/css; charset=utf-8',
  '.json': 'application/json; charset=utf-8',
  '.svg': 'image/svg+xml',
};

export function createHttpServer({ projectRoot, webDir }) {
  const clients = new Set();

  const server = http.createServer(async (req, res) => {
    const url = new URL(req.url, 'http://localhost');
    const pathname = decodeURIComponent(url.pathname);

    try {
      if (pathname === '/api/model') {
        const model = await loadProject(projectRoot);
        return sendJson(res, 200, model);
      }

      if (pathname === '/events') {
        res.writeHead(200, {
          'Content-Type': 'text/event-stream; charset=utf-8',
          'Cache-Control': 'no-cache',
          Connection: 'keep-alive',
        });
        res.write('event: ready\ndata: {}\n\n');
        clients.add(res);
        req.on('close', () => clients.delete(res));
        return;
      }

      if (!webDir) return sendJson(res, 404, { error: 'not found' });

      const rel = pathname === '/' ? 'index.html' : pathname.replace(/^\/+/, '');
      const abs = path.join(webDir, rel);
      const rootAbs = path.resolve(webDir);
      if (!path.resolve(abs).startsWith(rootAbs + path.sep) && path.resolve(abs) !== rootAbs) {
        return sendJson(res, 403, { error: 'forbidden' });
      }
      const data = await fs.readFile(abs);
      res.writeHead(200, { 'Content-Type': MIME[path.extname(abs)] ?? 'application/octet-stream' });
      return res.end(data);
    } catch (err) {
      if (err.code === 'ENOENT') return sendJson(res, 404, { error: 'not found' });
      return sendJson(res, 500, { error: String(err.message ?? err) });
    }
  });

  // 供 watcher 调用：向所有 SSE 客户端广播
  server.broadcastUpdate = () => {
    for (const res of clients) {
      try { res.write('event: update\ndata: {}\n\n'); } catch { clients.delete(res); }
    }
  };

  return server;
}

function sendJson(res, status, body) {
  const payload = JSON.stringify(body);
  res.writeHead(status, {
    'Content-Type': 'application/json; charset=utf-8',
    'Cache-Control': 'no-store',
  });
  res.end(payload);
}
```

```js
#!/usr/bin/env node
import path from 'node:path';
import { spawn } from 'node:child_process';
import { fileURLToPath } from 'node:url';
import { createHttpServer } from '../src/server/server.mjs';
import { createWatcher } from '../src/server/watcher.mjs';

export function parseArgs(argv) {
  const out = { project: process.cwd(), port: 0, open: false, help: false };
  for (let i = 0; i < argv.length; i++) {
    const a = argv[i];
    if (a === '--help' || a === '-h') out.help = true;
    else if (a === '--open') out.open = true;
    else if (a === '--project') out.project = path.resolve(argv[++i] ?? '.');
    else if (a === '--port') out.port = Number(argv[++i] ?? 0);
  }
  return out;
}

const USAGE = `用法: dayu-viz [--project <目录>] [--port <端口>] [--open]

  --project <目录>   含 docs/ 的 dayu 项目根目录（默认：当前目录）
  --port <端口>      监听端口（默认 0 = 随机空闲端口）
  --open             启动后用默认浏览器打开
`;

export async function main(argv = process.argv.slice(2)) {
  const args = parseArgs(argv);
  if (args.help) {
    process.stdout.write(USAGE);
    return;
  }

  const webDir = path.join(path.dirname(fileURLToPath(import.meta.url)), '..', 'web');
  const server = createHttpServer({ projectRoot: args.project, webDir });

  const watcher = createWatcher(
    [path.join(args.project, 'docs')],
    () => server.broadcastUpdate?.(),
    { debounceMs: 300 },
  );

  await new Promise((r) => server.listen(args.port, '127.0.0.1', r));
  const { port } = server.address();
  const url = `http://127.0.0.1:${port}/`;
  process.stdout.write(`dayu-viz 已启动：${url}\n项目：${args.project}\n按 Ctrl+C 退出\n`);

  if (args.open) {
    const cmd = process.platform === 'win32' ? 'cmd' : process.platform === 'darwin' ? 'open' : 'xdg-open';
    const cmdArgs = process.platform === 'win32' ? ['/c', 'start', '', url] : [url];
    try { spawn(cmd, cmdArgs, { stdio: 'ignore', detached: true }).unref(); } catch { /* 打开失败不影响服务 */ }
  }

  const shutdown = () => { watcher.close(); server.close(() => process.exit(0)); };
  process.on('SIGINT', shutdown);
  process.on('SIGTERM', shutdown);
}

if (process.argv[1] && import.meta.url.endsWith(path.basename(process.argv[1]))) {
  main().catch((err) => { process.stderr.write(String(err.stack ?? err) + '\n'); process.exit(1); });
}
```

```js
const UNKNOWN_TEXT = '未提供';

function todoList(items, emptyText) {

export function renderOverview(model) {
```

```js
import { renderOverview } from './views/overview.mjs';

const app = document.getElementById('app');
const conn = document.getElementById('conn');

export async function fetchModel(base = '') {
  const res = await fetch(`${base}/api/model`, { cache: 'no-store' });
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json();
}

export function subscribe(base, onUpdate) {
  const es = new EventSource(`${base}/events`);
  es.addEventListener('update', () => onUpdate());
  es.addEventListener('open', () => { conn.textContent = '实时跟随中'; });
  es.addEventListener('error', () => { conn.textContent = '连接断开，重试中…'; });
  return es;
}

async function refresh() {
  try {
    const model = await fetchModel();
    app.innerHTML = renderOverview(model);
  } catch (err) {
    app.innerHTML = `<p class="error">加载失败：${String(err.message ?? err)}</p>`;
  }
}

await refresh();
subscribe('', refresh);
```
